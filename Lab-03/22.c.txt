#include <stdio.h>
void main()
{
    // WAP to print Pascal triangle
    int n;
    printf("enter n:");
    scanf("%d", &n);
    int a[n][n], i, j;
    for (i = 1; i <=n; i++)
    {
        for (j = 1; j <= i; j++)
        {
            if (j == i || j == 1)
            {
                a[i][j] = 1;
            }
            else
            {
                a[i][j] = a[i - 1][j - 1] + a[i - 1][j];
            }
        }
    }
    for (i = 1; i <=n; i++)
    {
        for (j = 1; j <=n - i; j++)
        {
            printf(" ");
        }
        for (j = 1; j <= i; j++)
        {
            printf("%d", a[i][j]);
        }
        printf("\n");
    }
}
