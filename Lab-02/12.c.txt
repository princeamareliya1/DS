#include<stdio.h>
void main(){
    // WAP to check whether a number is prime or not.
    int n,i,s=0;
    printf("enter no:");
    scanf("%d",&n);
    for ( i =2; i <n; i++)
    {
        if(n%i==0)
        {
            s++;
        
        }
    }
    if(s==0)
    {
        printf("prime no");
    }else{
        printf("NOT A PRIME NO");
    }
    
}
