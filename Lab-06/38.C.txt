#include<stdio.h>
#include<stdlib.h>
int main()
{
    // WAP to sort the array elements using Pointer(BUBBLE SORT)
    int n,i,j,t;
    printf("Enter NO:");
    scanf("%d",&n);
    int *a = (int *)malloc(n*sizeof(int));
    for ( i = 0; i <n; i++)
    {
        printf("ENTER ELEMENT OF:");
        scanf("%d",&a[i]);
    }
        for ( i = 0; i <n; i++)
    {
        printf("%d\t",a[i]);
    }
    printf("\n");
    for ( i = 0; i <n-1; i++)
    {
        for ( j = 0; j <n-i-1; j++)
        {
            if (a[j]>a[j+1])
            {
               t=a[j+1];
               a[j+1]=a[j];
               a[j]=t; 
            }
                
        }
        

    }
    printf("AFTER SORTING:\t");
    for ( i = 0; i <n; i++)
    {
        printf("%d\t",a[i]);
    }
    free(a);
    
    return 0;
}
